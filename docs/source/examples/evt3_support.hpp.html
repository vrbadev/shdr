118/** @brief Pack decoded CD events into X11/Y10/polarity1/time42 uint64 records.
119 * This CPU consumer downloads completed GPU results and packs CD pixel events.
120 * The lossless layout accepts X11, Y10 and time42 values, with range errors
121 * reported as exceptions. The original uvec4 records support larger geometry.
122 */

124evt3_compact_cd_to64(const shdr::buffer<shdr::glsl::uvec4>& decoded, unsigned count) { // Apply evt3 compact cd to64 to these arguments.
125    if (count > decoded.size()) // Check this condition before entering the following branch.
126        throw shdr::error("EVT3 compact count exceeds decoded storage"); // Report EVT3 compact count exceeds decoded storage.
127    const shdr::glsl::uvec4* events = decoded.data(); // Set events from this expression.
128    std::vector<std::uint64_t> packed; // Declare packed for the following calculation.
129    packed.reserve(count); // Apply reserve to these arguments.
130    for (unsigned i = 0; i < count; ++i) { // Iterate over i for this calculation.
131        const shdr::glsl::uvec4& event = events[i]; // Set event from this expression.
132        if (event.y & 0x80000000u) // Check this condition before entering the following branch.
133            continue; // Skip the remaining work for this loop iteration.
134        const unsigned y = event.y & 2047u; // Set y from this expression.
135        if (event.x >= 2048u || y >= 1024u) // Check this condition before entering the following branch.
136            throw shdr::error("64-bit EVT3 CD packing requires X<2048 and Y<1024"); // Report 64-bit EVT3 CD packing requires X<2048 and Y<1024.
137        if (event.w >= 1024u) // Check this condition before entering the following branch.
138            throw shdr::error("64-bit EVT3 CD packing requires timestamps below 2^42 microseconds"); // Report 64-bit EVT3 CD packing requires timestamps below 2^42 microseconds.
139        const std::uint64_t timestamp = (std::uint64_t(event.w) << 32u) | event.z; // Set timestamp from this expression.
140        const unsigned polarity = (event.y >> 11u) & 1u; // Set polarity from this expression.
141        packed.push_back(std::uint64_t(event.x) | (std::uint64_t(y) << 11u) | // Apply push back to these arguments.
142                         (std::uint64_t(polarity) << 21u) | (timestamp << 22u)); // Apply uint64 t to these arguments.
143    } // End this block.
