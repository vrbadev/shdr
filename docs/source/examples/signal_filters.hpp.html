32inline std::vector<double> low_pass(double normalized, unsigned taps) { // Build a Hamming-windowed sinc with unit DC gain.
33    std::vector<double> result(taps); // Keep filter design coefficients in double precision until normalization.
34    const int center = static_cast<int>(taps / 2u); // Find the symmetry center of the odd-length FIR.
35    double sum = 0.0; // Start the scalar accumulator at zero.
36    for (unsigned i = 0; i < taps; ++i) { // Use only causal taps that lie inside this batch.
37        const int offset = static_cast<int>(i) - center; // Measure this tap relative to the symmetry center.
38        const double sinc = offset ? std::sin(2.0 * pi() * normalized * offset) / (pi() * offset) // Evaluate the ideal low-pass sinc around the center tap.
39                                   : 2.0 * normalized; // Use the finite sinc limit at its center tap.
40        const double window = 0.54 - 0.46 * std::cos(2.0 * pi() * i / (taps - 1u)); // Apply the Hamming window to reduce sidelobes.
41        sum += result[i] = sinc * window; // Store the windowed tap and accumulate its DC gain.
42    } // Finish this channel, sample or feature pass.
43    if (!std::isfinite(sum) || std::fabs(sum) < 1e-12) // Reject a nonfinite or out-of-range value.
44        throw shdr::error("FIR design has negligible normalization gain"); // Report the invalid design or failed check clearly.
45    for (double& value : result) // Apply the design adjustment to every coefficient.
46        value /= sum; // Normalize this tap by the computed passband gain.
47    return result; // Return the finite designed filter coefficients.
48} // Finish the low_pass operation.
49inline void normalize(std::vector<double>& values, double normalized) { // Set the requested passband gain to one.
50    double gain = 0.0; // Start the scalar accumulator at zero.
51    const int center = static_cast<int>(values.size() / 2u); // Find the symmetry center of the odd-length FIR.
52    for (unsigned i = 0; i < values.size(); ++i) // Visit every value needed by this calculation.
53        gain += values[i] * std::cos(2.0 * pi() * normalized * (static_cast<int>(i) - center)); // Sum the zero-phase response at the selected frequency.
54    if (!std::isfinite(gain) || std::fabs(gain) < 1e-12) // Reject a nonfinite or out-of-range value.
55        throw shdr::error("FIR design has negligible normalization gain"); // Report the invalid design or failed check clearly.
56    for (double& value : values) // Apply the design adjustment to every coefficient.
57        value /= gain; // Normalize this tap by the computed passband gain.
58} // Finish the normalize operation.
59inline void validate_biquad(const std::array<float, 5>& coefficients) { // Reject nonfinite or unstable float coefficients.

66} // Finish the validate_biquad operation.

86/** @brief Design an odd-length linear-phase FIR using a Hamming-windowed sinc.
87 * Low-pass coefficients have unit DC gain. High-pass uses spectral inversion
88 * and unit Nyquist gain. Band-pass subtracts two low-pass designs and has unit
89 * midpoint gain.  band-stop inverts that band-pass and has unit DC gain.
90 * Moving-average returns equal weights and ignores both cutoffs. Other modes
91 * require cutoffs strictly inside Nyquist.  band edges must increase. Tap count
92 * must be odd and between 3 and 4095. The causal group delay is (taps-1)/2.
93 */
94inline std::vector<float> fir_coefficients(filter_mode mode, // Design the requested causal linear-phase FIR.
95                                           double sample_rate, // Receive the positive sampling rate in frequency units.
96                                           double cutoff1, // Receive the cutoff or band-edge frequency.
97                                           double cutoff2 = 0.0, // Receive the cutoff or band-edge frequency.
98                                           unsigned taps = 31u) { // Receive the validated sample, channel or coefficient count.
99    signal_detail::sample_rate(sample_rate); // Validate the sample-rate units before designing coefficients.
100    if (taps < 3u || taps > 4095u || taps % 2u == 0u) // Handle this validation or boundary case explicitly.
101        throw shdr::error("FIR tap count must be odd and between 3 and 4095"); // Report the invalid design or failed check clearly.
102    if (mode == filter_mode::moving_average) // Choose the coefficient rule for this filter shape.
103        return std::vector<float>(taps, 1.0f / taps); // Give every moving-average tap the same weight.
104    signal_detail::cutoff(cutoff1, sample_rate); // Validate this frequency against the sampling rate.
105    std::vector<double> values; // Keep filter design coefficients in double precision until normalization.
106    if (mode == filter_mode::low_pass || mode == filter_mode::high_pass) { // Choose the coefficient rule for this filter shape.
107        values = signal_detail::low_pass(cutoff1 / sample_rate, taps); // Build the appropriate low-pass prototype.
108        if (mode == filter_mode::high_pass) { // Choose the coefficient rule for this filter shape.
109            for (double& value : values) // Apply the design adjustment to every coefficient.
110                value = -value; // Invert this coefficient for spectral inversion.
111            values[taps / 2u] += 1.0; // Add the centered impulse to complete spectral inversion.
112            signal_detail::normalize(values, 0.5); // Scale the chosen passband response to unity.
113        } // Finish handling this condition.
114    } else if (mode == filter_mode::band_pass || mode == filter_mode::band_stop) { // Use the alternative filter-design case.
115        signal_detail::cutoff(cutoff2, sample_rate); // Validate this frequency against the sampling rate.
116        if (cutoff1 >= cutoff2) // Require the lower band edge to precede the upper edge.
117            throw shdr::error("FIR band edges must be strictly increasing"); // Report the invalid design or failed check clearly.
118        const std::vector<double> lower = signal_detail::low_pass(cutoff1 / sample_rate, taps); // Keep the lower-edge low-pass prototype for subtraction.
119        values = signal_detail::low_pass(cutoff2 / sample_rate, taps); // Build the appropriate low-pass prototype.
120        for (unsigned i = 0; i < taps; ++i) // Use only causal taps that lie inside this batch.
121            values[i] -= lower[i]; // Subtract the lower low-pass response to form a band-pass.
122        signal_detail::normalize(values, 0.5 * (cutoff1 / sample_rate + cutoff2 / sample_rate)); // Scale the chosen passband response to unity.
123        if (mode == filter_mode::band_stop) { // Choose the coefficient rule for this filter shape.
124            for (double& value : values) // Apply the design adjustment to every coefficient.
125                value = -value; // Invert this coefficient for spectral inversion.
126            values[taps / 2u] += 1.0; // Add the centered impulse to complete spectral inversion.
127            signal_detail::normalize(values, 0.0); // Scale the chosen passband response to unity.
128        } // Finish handling this condition.
129    } else // Use the alternative filter-design case.
130        throw shdr::error("unknown FIR filter mode"); // Report the invalid design or failed check clearly.
131    std::vector<float> result(values.begin(), values.end()); // Convert the designed coefficients to GPU-compatible floats.
132    for (float value : result) // Inspect each stored float coefficient.
133        if (!std::isfinite(value)) // Reject a nonfinite or out-of-range value.
134            throw shdr::error("FIR coefficients must remain finite at float precision"); // Report the invalid design or failed check clearly.
135    return result; // Return the finite designed filter coefficients.
136} // Finish handling this condition.

138/** @brief Design a normalized RBJ/W3C low-pass, high-pass, band-pass or notch biquad.
139 * Return order is b0,b1,b2,a1,a2 with a0 normalized to one. Band-pass has unit
140 * center-frequency gain.  band_stop selects a notch. Frequency is the cutoff or
141 * center in sample-rate units, and Q must be finite and positive. Designs whose
142 * rounded float feedback coefficients lose strict stability are rejected.
143 * Formula reference: https://www.w3.org/TR/audio-eq-cookbook/ .
144 */
145inline std::array<float, 5> biquad_coefficients(filter_mode mode, // Design normalized coefficients from the RBJ formulas.
146                                                double sample_rate, // Receive the positive sampling rate in frequency units.
147                                                double frequency, // Receive the tone, cutoff or center frequency.
148                                                double q = 0.7071067811865475244) { // Receive the positive biquad quality factor.
149    signal_detail::sample_rate(sample_rate); // Validate the sample-rate units before designing coefficients.
150    signal_detail::cutoff(frequency, sample_rate); // Validate this frequency against the sampling rate.
151    if (!std::isfinite(q) || q <= 0.0) // Reject a nonfinite or out-of-range value.
152        throw shdr::error("biquad Q must be finite and positive"); // Report the invalid design or failed check clearly.
153    const double omega = 2.0 * signal_detail::pi() * (frequency / sample_rate); // Convert the cutoff frequency into radians per sample.
154    const double cosine = std::cos(omega), alpha = std::sin(omega) / (2.0 * q), a0 = 1.0 + alpha; // Compute the RBJ frequency and damping terms.
155    double b0 = 0.0, b1 = 0.0, b2 = 0.0; // Prepare the three feed-forward numerator coefficients.
156    if (mode == filter_mode::low_pass) { // Choose the coefficient rule for this filter shape.
157        b0 = b2 = (1.0 - cosine) / 2.0; // Use symmetric outer numerator coefficients for this filter.
158        b1 = 1.0 - cosine; // Set the middle numerator coefficient for this filter.
159    } else if (mode == filter_mode::high_pass) { // Use the alternative filter-design case.
160        b0 = b2 = (1.0 + cosine) / 2.0; // Use symmetric outer numerator coefficients for this filter.
161        b1 = -(1.0 + cosine); // Set the middle numerator coefficient for this filter.
162    } else if (mode == filter_mode::band_pass) { // Use the alternative filter-design case.
163        b0 = alpha; // Give the band-pass unit gain at its center frequency.
164        b2 = -alpha; // Use the opposite outer coefficient for the band-pass zero.
165    } else if (mode == filter_mode::band_stop) { // Use the alternative filter-design case.
166        b0 = b2 = 1.0; // Use symmetric outer numerator coefficients for this filter.
167        b1 = -2.0 * cosine; // Set the middle numerator coefficient for this filter.
168    } else // Use the alternative filter-design case.
169        throw shdr::error("biquad supports low-pass, high-pass, band-pass and band-stop modes"); // Report the invalid design or failed check clearly.
170    const std::array<float, 5> result = {{static_cast<float>(b0 / a0), // Normalize every coefficient by a0 before storing floats.
171                                          static_cast<float>(b1 / a0), // Convert the computed host value to the shader float type.
172                                          static_cast<float>(b2 / a0), // Convert the computed host value to the shader float type.
173                                          static_cast<float>(-2.0 * cosine / a0), // Convert the computed host value to the shader float type.
174                                          static_cast<float>((1.0 - alpha) / a0)}}; // Convert the computed host value to the shader float type.
175    signal_detail::validate_biquad(result); // Require stable finite feedback coefficients.
176    return result; // Return the finite designed filter coefficients.
177} // Finish handling this condition.
