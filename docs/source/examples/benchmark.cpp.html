468void async_pipeline(benchmark& bench, shdr::device* device) { // Implement async pipeline for the operations below.
469    const unsigned count = 257, frame_count = 32, slot_count = 3; // Set count from this expression.
470    struct slot { // Declare slot.
471        shdr::buffer<float> input, output; // Declare input for the following calculation.
472        std::future<void> cpu_done; // Declare cpu done for the following calculation.
473        shdr::submission gpu_done; // Declare gpu done for the following calculation.
474        bool occupied; // Declare occupied for the following calculation.
475        unsigned frame; // Declare frame for the following calculation.
476        slot() : input(count), output(count), occupied(false), frame(0) { // Apply slot to these arguments.
477        } // End this block.
478    }; // Finish this declaration.
479    std::vector<slot> slots(slot_count); // Allocate three disjoint input and output slots before measuring the ring.
480    for (unsigned i = 0; i < slot_count; ++i) // Iterate over i for this calculation.
481        shdr_examples::randomize(slots[i].input, random_seed + i, -1.0f, 1.0f); // Fill the input with reproducible random values.
482    auto shader = shdr::make_kernel("affine_kernel"); // Create the linked affine kernel instance.
483    shader->set<unsigned>("count", count); // Supply this shader uniform with its typed application value.
484    shdr::pipeline pipeline = shdr::dispatch(*shader, groups(*shader, count)); // Add a shader dispatch with the required workgroup counts.
485    shdr::executor workers(2, slot_count); // Use two CPU workers with enough bounded queue capacity for the three live slots.
486    if (device) { // Check this condition before entering the following branch.
487        for (slot& resource : slots) { // Visit each resource required by this loop.
488            shader->bind("input_values", resource.input).bind("output_values", resource.output); // Connect this shader port to its application resource.
489            pipeline.prepare(*device); // Prepare all slot resources and compile the GPU program before ring timing.
490        } // End this block.
491    } // End this block.
492    const auto execute = [&](bool gpu, bool validate) { // Set execute from this expression.
493        unsigned produced = 0, completed = 0; // Set produced from this expression.
494        while (completed < frame_count) { // Pump the three resource slots until the complete frame batch finishes.
495            bool progress = false; // Set progress from this expression.
496            for (unsigned index = 0; index < slot_count; ++index) { // Iterate over index for this calculation.
497                slot& resource = slots[index]; // Set resource from this expression.
498                if (resource.occupied) { // Check this condition before entering the following branch.
499                    const bool ready = // Set ready from this expression.
500                        gpu ? resource.gpu_done.ready() // Apply ready to these arguments.

503                    if (!ready) // Keep this slot occupied until its preceding operation actually finishes.

505                    if (!gpu) // Check this condition before entering the following branch.
506                        resource.cpu_done.get(); // Observe CPU worker exceptions before considering this slot reusable.
507                    if (validate) { // Check every completed warmup frame while keeping readback outside the timed ring.
508                        const shdr::buffer<float>& input = resource.input; // Use a constant buffer view so GPU writes synchronize before host reads.
509                        const shdr::buffer<float>& output = resource.output; // Use a constant buffer view so GPU writes synchronize before host reads.
510                        for (unsigned i = 0; i < count; ++i) // Iterate over i for this calculation.
511                            same(output.data()[i], // Compare CPU and GPU values using the selected tolerance.
512                                 input.data()[i] * float(resource.frame + 1) - // Apply data to these arguments.
513                                     float(resource.frame), // Apply float to these arguments.
514                                 1e-4, // Complete this expression or declaration.
515                                 1e-4); // Complete this expression or declaration.
516                    } // End this block.
517                    resource.occupied = false; // Release this slot only after its prior frame has completed.
518                    ++completed; // Advance the completed-frame counter after observing its fence or future.
519                    progress = true; // Set progress from this expression.
520                } // End this block.
521                if (produced < frame_count && produced % slot_count == index) { // Assign frames round-robin so both backends receive the same slot inputs.
522                    resource.frame = produced++; // Record the frame identity before changing its shader uniforms.
523                    shader->bind("input_values", resource.input) // Connect this shader port to its application resource.
524                        .bind("output_values", resource.output) // Connect this shader port to its application resource.
525                        .set<float>("scale", float(resource.frame + 1)) // Supply this shader uniform with its typed application value.

539        } // End this block.
