33inline double fir_gain(const std::vector<float>& coefficients, double frequency, double rate) { // Evaluate FIR gain using complex frequency response.
34    std::complex<double> gain(0, 0), phase(1, 0), step = unit_phase(frequency, rate); // Keep the complex phase and accumulated transfer response.
35    for (float coefficient : coefficients) { // Visit every value needed by this calculation.
36        gain += double(coefficient) * phase; // Add this tap at its complex frequency phase.
37        phase *= step; // Advance to the next tap delay.
38    } // Finish this channel, sample or feature pass.
39    return std::abs(gain); // Return the transfer-function magnitude.
40} // Finish the fir_gain operation.
41inline double biquad_gain(const std::array<float, 5>& coefficients, double frequency, double rate) { // Evaluate the normalized biquad transfer function.
42    const std::complex<double> phase = unit_phase(frequency, rate); // Keep the complex phase and accumulated transfer response.
43    return std::abs( // Return the transfer-function magnitude.
44        (double(coefficients[0]) + double(coefficients[1]) * phase + // Evaluate the feed-forward numerator at this frequency.
45         double(coefficients[2]) * phase * phase) / // Include the second delayed feed-forward coefficient.
46        (1.0 + double(coefficients[3]) * phase + double(coefficients[4]) * phase * phase)); // Divide by the normalized feedback denominator.
47} // Finish the biquad_gain operation.

73inline std::vector<double> biquad_reference(const std::vector<float>& input, // Use a direct-form I double recurrence as an independent oracle.
74                                            unsigned count, // Receive the validated sample, channel or coefficient count.
75                                            unsigned channels, // Receive the validated sample, channel or coefficient count.
76                                            const std::array<float, 5>& coefficients, // Use five normalized biquad coefficients.
77                                            std::vector<shdr::glsl::vec2>* final_state = 0) { // Keep the independent expected recurrence pair for each channel.
78    std::vector<double> output(input.size()); // Allocate independent double oracle outputs.
79    if (final_state) // Record reference state only when the caller asks for it.
80        final_state->resize(channels); // Reserve one final recurrence pair per channel.
81    for (unsigned channel = 0; channel < channels; ++channel) { // Process each channel independently.
82        double x1 = 0.0, x2 = 0.0, y1 = 0.0, y2 = 0.0; // Start the independent direct-form I recurrence with zero history.
83        for (unsigned i = 0; i < count; ++i) { // Visit every value needed by this calculation.
84            const double x = input[channel * count + i]; // Read this input sample for the double recurrence.
85            const double y = coefficients[0] * x + coefficients[1] * x1 + coefficients[2] * x2 - // Apply the independent direct-form I difference equation.
86                             coefficients[3] * y1 - coefficients[4] * y2; // Subtract the two normalized feedback contributions.
87            output[channel * count + i] = y; // Keep this independently computed filtered sample.
88            x2 = x1; // Shift the input history by one sample.
89            x1 = x; // Remember the current input for the next sample.
90            y2 = y1; // Shift the output history by one sample.
91            y1 = y; // Remember the current output for the next sample.
92        } // Finish this channel, sample or feature pass.
93        if (final_state) // Record reference state only when the caller asks for it.
94            (*final_state)[channel] = // Convert direct-form I history into transposed recurrence state.
95                shdr::glsl::vec2(float(coefficients[1] * x1 + coefficients[2] * x2 - // Convert the last input and output histories into the first state value.
96                                       coefficients[3] * y1 - coefficients[4] * y2), // Include the feedback history in the first state value.
97                                 float(coefficients[2] * x1 - coefficients[4] * y1)); // Convert the double reference value for the float comparison.
98    } // Finish this channel, sample or feature pass.
99    return output; // Return every independently computed filtered sample.
100} // Finish the biquad_reference operation.

109inline double tone_amplitude(const shdr::buffer<float>& samples, // Measure a tone by sine and cosine projection.
