19float logistic(float value) { // Evaluate the sigmoid with a sign-selected, numerically stable exponential.
20    if (value >= 0.0) // Use the negative exponent when the input logit is nonnegative.
21        return 1.0 / (1.0 + exp(-value)); // Convert a nonnegative logit into its sigmoid probability.
22    float e = exp(value); // Use a nonpositive exponent for the negative logit branch.
23    return e / (1.0 + e); // Convert a negative logit into its sigmoid probability.
24} // End this block.
25void main() { // Differentiate one sample loss with respect to one ANN parameter.
26    uint index = gl_GlobalInvocationID.x, row = index / 33u, parameter = index % 33u; // Select one sample-major derivative among all 33 ANN parameters.
27    if (row >= batch_count) // Ignore invocations outside the sample batch.
28        return; // Stop this invocation before performing another resource access.
29    float x = inputs[2u * row], y = inputs[2u * row + 1u], logit = parameters[32]; // Read the two input features and initialize the output with its bias.
30    float activations[8], preactivations[8]; // Retain hidden preactivations and ReLU outputs for the chain rule.
31    for (uint neuron = 0u; neuron < 8u; ++neuron) { // Evaluate each of the eight hidden neurons.
32        preactivations[neuron] = parameters[2u * neuron] * x + parameters[2u * neuron + 1u] * y + // Multiply the two input features by this neuron's learned weights.
33                                 parameters[16u + neuron]; // Include this hidden neuron's bias before the ReLU.
34        activations[neuron] = max(0.0, preactivations[neuron]); // Form the hidden neuron's weighted input before its nonlinearity.
35        logit += activations[neuron] * parameters[24u + neuron]; // Keep positive preactivations and set the negative branch to zero.
36    } // End this block.
37    float delta = logistic(logit) - targets[row], gradient = delta; // Use prediction minus target as the cross-entropy derivative with respect to the logit.
38    if (parameter < 16u) { // Differentiate one of the sixteen hidden input weights.
39        uint neuron = parameter / 2u; // Find the hidden neuron owning this input weight.
40        gradient = preactivations[neuron] > 0.0 // Form the hidden neuron's weighted input before its nonlinearity.
41                       ? delta * parameters[24u + neuron] * (parameter % 2u == 0u ? x : y) // Multiply the output derivative by its output weight and the selected input feature.
42                       : 0.0; // Use zero derivative for the inactive ReLU branch.
43    } else if (parameter < 24u) { // Differentiate one of the eight hidden biases.
44        uint neuron = parameter - 16u; // Find the hidden neuron owning this bias.
45        gradient = preactivations[neuron] > 0.0 ? delta * parameters[24u + neuron] : 0.0; // Form the hidden neuron's weighted input before its nonlinearity.
46    } else if (parameter < 32u) // Differentiate one of the eight output weights.
47        gradient = delta * activations[parameter - 24u]; // Multiply the output logit derivative by this hidden activation.
48    gradients[index] = gradient; // Store this sample's derivative for the following batch SGD stage.
49} // End this block.
