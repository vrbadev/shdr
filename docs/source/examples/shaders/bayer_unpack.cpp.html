16uint byte_at(uint position) { // Extract one byte from the little-endian packed uint buffer.
17    return (packed_words[position / 4u] >> ((position % 4u) * 8u)) & 255u; // Shift the requested byte to the low bits and mask away the other three bytes.
18} // End this block.
19void main() { // Decode one packed sensor sample into an unsigned Bayer value.
20    uint i = gl_GlobalInvocationID.x; // Select this invocation's global x coordinate.
21    if (i >= count) // Stop padding invocations before reading or writing an element.
22        return; // Stop this invocation before performing another resource access.
23    uint value = 0u; // Initialize the unpacked sample before selecting its byte format.
24    if (packing == 0u) // Decode a format with one complete sample in each byte.
25        value = byte_at(i); // Read the Bayer8 sample directly.
26    else if (packing == 1u) // Decode a little-endian format with two bytes per sample.
27        value = byte_at(i * 2u) | (byte_at(i * 2u + 1u) << 8u); // Combine the low and high bytes of the Bayer16 sample.
28    else if (packing == 2u) { // Decode four RAW10 samples sharing one low-bit byte.
29        uint start = (i / 4u) * 5u; // Locate this four-sample RAW10 group of five bytes.
30        value = (byte_at(start + i % 4u) << 2u) | ((byte_at(start + 4u) >> ((i % 4u) * 2u)) & 3u); // Combine the eight high bits with this sample's two low bits.
31    } else { // Decode two RAW12 samples sharing one low-bit byte.
32        uint start = (i / 2u) * 3u; // Locate this two-sample RAW12 group of three bytes.
33        value = (byte_at(start + i % 2u) << 4u) | ((byte_at(start + 2u) >> ((i % 2u) * 4u)) & 15u); // Combine the eight high bits with this sample's four low bits.
34    } // End this block.
35    samples[i] = value & ((1u << bit_depth) - 1u); // Keep only the declared valid sensor bits in the unpacked sample.
36} // End this block.
