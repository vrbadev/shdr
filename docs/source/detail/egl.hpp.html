48        egl_display_registry& registry = egl_displays(); // Set the shared display lifetime registry.
49        std::lock_guard<std::mutex> lock(registry.mutex); // Hold the mutex while this shared state is accessed.
50        if (registry.owners) { // Reuse this display when another context already retains its initialization.
51            display = registry.display; // Set the initialized EGL display.
52            ++registry.owners; // Retain the existing initialized display for this new owner.
53            return; // Finish this operation.
54        } // End this scope in egl display state.

57            reinterpret_cast<platform_display_fn>(eglGetProcAddress("eglGetPlatformDisplayEXT")); // Look up the requested GL entry point through EGL.

63        } // End this scope in egl display state.

65            display = eglGetDisplay(EGL_DEFAULT_DISPLAY); // Set the initialized EGL display.
66            if (display == EGL_NO_DISPLAY || !eglInitialize(display, 0, 0)) { // Report failure to acquire and initialize the default display.
67                display = EGL_NO_DISPLAY; // Set the initialized EGL display.
68                throw egl_failure("EGL display"); // Report egl display.
69            } // End this scope in egl display state.
70        } // End this scope in egl display state.

159        if (eglGetCurrentContext() == context && eglGetCurrentDisplay() == display->display) // Choose this branch when eglGetCurrentContext() equals context and eglGetCurrentDisplay() equals display->display.
