44        if (!fence) // Finish immediately for an empty fence.
45            return; // Finish this operation.
46        const GLsync sync = fence; // Copy the fence handle into the deferred cleanup callback.
47        const std::shared_ptr<submission_resources> keep = retained; // Retain buffers and programs until deferred cleanup releases this bundle.
48        try { // Keep cleanup available if an allocation, shader, or driver operation fails.
49            if (completed) { // Retire a fence whose completion has already been observed.
50                owner->retire([sync, keep](gl_api& gl) { // Delete on the context thread or defer deletion to that thread.

52                    (void)keep; // Keep the resource references captured for this cleanup.
53                }); // End this scope in submission state.
54            } else { // Handle the alternative resource or completion case.
55 // The context thread checks this retirement entry with a zero timeout.
56                owner->defer_poll( // Retain a completion check until its fence signals.
57                    [sync, keep](gl_api& gl) -> bool { // Retain these captures until the queued callback finishes.
58                        const GLenum status = gl.ClientWaitSync(sync, 0, 0); // Poll the discarded fence with zero flags and a zero timeout.
59                        if (status == TIMEOUT_EXPIRED) // Check the driver result before marking GPU completion.
60                            return false; // Keep the discarded submission pending while its fence is unsignaled.
61                        gl.DeleteSync(sync); // Delete the GL completion fence.
62                        (void)keep; // Keep the resource references captured for this cleanup.
63                        return true; // Remove the retired fence and release its retained resource bundle.
64                    }, // End this scope in submission state.
65                    [sync](gl_api& gl) { // Keep only the fence handle for nonblocking shutdown cancellation.
66                        gl.DeleteSync(sync); // Delete the GL completion fence.
67                    }); // End this scope in submission state.
68            } // End this scope in submission state.
69        } catch (...) { // Preserve cleanup or exception propagation for this failed operation.
70 // Destruction defers pending fence cleanup. GL command references keep
71 // already queued operations valid if allocating a retirement entry fails.
72        } // End this scope in submission state.
73    } // End this scope in submission state.

75        owner->require_thread(); // Require the device context thread and activate its headless context.
76        if (completed) // Reuse the cached completion result after checking the context thread.
77            return wait_status::ready; // Tell the caller that queued GPU commands have completed.
78        const GLenum result = owner->gl.ClientWaitSync(fence, 0, timeout); // Check a GPU fence with the caller-selected timeout.

82        } // End this scope in query.

86        throw error("GPU completion query failed"); // Report gpu completion query failed.
87    } // End this scope in query.

140/** @brief Enqueue all GPU stages and return a fence for polling or explicit waits.

146inline submission pipeline::submit(device& d) const { // Queue compute work and return its completion handle.
147    d.state()->require_thread(); // Require the device context thread and activate its headless context.
148    const std::vector<stage>& commands = stages(); // Read the ordered stages that this submission will queue.
149    for (std::size_t i = 0; i < commands.size(); ++i) // Visit each ordered GPU stage in this submission.
150        if (commands[i].target == backend::cpu) // Reject explicit CPU stages before queueing any GPU commands.
151            throw error("GPU submission cannot contain CPU stages"); // Require GPU stages for GPU submission.
152    if (commands.empty()) // Skip fence creation for an empty pipeline.
153        return submission(); // Represent an empty pipeline with an already completed handle.
154    std::unique_ptr<detail::submission_state> completion(new detail::submission_state(d.state())); // Retain the device, fence, and queued resource bundle.
155    completion->retained->programs.reserve(commands.size()); // Reserve room to retain one compiled program per stage.
156    prepare(d); // Compile the program and prepare its bound GPU resources.
157    for (std::size_t i = 0; i < commands.size(); ++i) { // Visit each ordered GPU stage in this submission.
158        const std::vector<std::shared_ptr<detail::resource_state>> resources = // Set the resource storage retained until queued work completes.
159            commands[i].shader->resources(); // Collect the shared storage used by this shader.
160        completion->retained->resources.insert( // Retain this entry in the ordered resource or work collection.
161            completion->retained->resources.end(), resources.begin(), resources.end()); // Keep the end of the current range.
162        completion->retained->programs.push_back(commands[i].shader->program_state()); // Retain this entry in the ordered resource or work collection.
163    } // End this scope in submit.
164    try { // Keep cleanup available if an allocation, shader, or driver operation fails.
165        run(d); // Queue all validated GPU stages in pipeline order.
166    } catch (...) { // Preserve cleanup or exception propagation for this failed operation.
167 // A later dispatch can fail after an earlier dispatch was queued.
168        completion->fence = d.state()->gl.FenceSync(detail::SYNC_GPU_COMMANDS_COMPLETE, 0); // Insert a completion fence after queued compute work.
169        d.state()->gl.Flush(); // Flush queued GPU commands for driver execution.
170        throw; // Rethrow the original failure after arranging the necessary cleanup.
171    } // End this scope in submit.
172    completion->fence = d.state()->gl.FenceSync(detail::SYNC_GPU_COMMANDS_COMPLETE, 0); // Insert a completion fence after queued compute work.
173    if (!completion->fence) { // Check that the driver allocated a valid completion fence.
174        d.state()->gl.check("create GPU submission fence"); // Expose retained runtime state to the resource and backend helpers.
175        throw error("cannot create GPU submission fence"); // Report completion-fence allocation failure.
176    } // End this scope in submit.
177    d.state()->gl.Flush(); // Flush queued GPU commands for driver execution.
178    d.state()->gl.check("submit GPU pipeline"); // Expose retained runtime state to the resource and backend helpers.
179    return submission(std::move(completion)); // Queue compute work and return its completion handle.
