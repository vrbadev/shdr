20def read_idx(path, dimensions):  # Validate and decode one official MNIST IDX archive.
21    raw = gzip.decompress(path.read_bytes())  # Expand only the named dataset archive.
22    magic = struct.unpack_from(">I", raw)[0]  # Read the IDX type and dimension tag.
23    if magic != 0x0800 + dimensions:  # Accept only unsigned-byte images or labels.
24        raise ValueError(f"unexpected IDX header in {path.name}")  # Reject another data format.
25    shape = struct.unpack_from(">" + "I" * dimensions, raw, 4)  # Read every dimension.
26    offset = 4 + 4 * dimensions  # Locate the first payload byte after the header.
27    size = 1  # Begin the expected payload-size product.
28    for dimension in shape:  # Include every declared array dimension.
29        size *= dimension  # Count the unsigned-byte values in the archive.
30    if len(raw) != offset + size:  # Check both truncation and trailing unexpected content.
31        raise ValueError(f"IDX payload size differs in {path.name}")  # Report unsafe input.
32    return raw[offset:], shape  # Return verified payload bytes and their shape.

54    class LeNet(nn.Module):  # Name the modern LeNet variant implemented by the native shaders.
55        def __init__(self):  # Declare layers in exported parameter order.
56            super().__init__()  # Initialize the framework's ordinary layer bookkeeping.
57            self.conv1 = nn.Conv2d(1, 6, 5)  # Map padded input to six 28 by 28 feature maps.
58            self.conv2 = nn.Conv2d(6, 16, 5)  # Map pooled features to sixteen 10 by 10 maps.
59            self.dense1 = nn.Linear(400, 120)  # Apply the full C5-equivalent dense layer.
60            self.dense2 = nn.Linear(120, 84)  # Apply the second hidden classifier layer.
61            self.dense3 = nn.Linear(84, 10)  # Produce one logit per handwritten digit class.
62
63        def forward(self, values):  # Match the native seven-dispatch inference pipeline.
64            values = nn.functional.pad(values, (2, 2, 2, 2))  # Add a zero border to 32 by 32.
65            values = nn.functional.avg_pool2d(torch.relu(self.conv1(values)), 2)  # First conv and pool.
66            values = nn.functional.avg_pool2d(torch.relu(self.conv2(values)), 2)  # Second conv and pool.
67            values = torch.relu(self.dense1(values.flatten(1)))  # Flatten channel-major pooled maps.
68            values = torch.relu(self.dense2(values))  # Apply the second hidden ReLU layer.
69            return self.dense3(values)  # Keep raw logits for stable cross entropy and comparison.
