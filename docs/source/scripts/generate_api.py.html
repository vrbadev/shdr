30def clean_comment(raw: str) -> str:  # Remove C++ comment prefixes before rendering their text.

38def declaration_after(source: str, end: int) -> str:  # Read the C++ declaration immediately following a docstring.
39    """Read a declaration, omitting inline bodies and multiline macro bodies."""
40    remainder = source[end:].lstrip()  # Read the text following this declaration or field.
41    if not remainder or remainder.startswith(("/**", "///")):  # Select comments attached to a declaration.
42        return ""  # Return an empty signature for this unattached comment.
43    if remainder.startswith("#define"):  # Show the macro name and parameters as its signature.
44        macro = re.match(r"#define\s+\w+(?:\([^)]*\))?", remainder)  # Recognize a generated macro or declaration name.
45        return macro.group() if macro else remainder.splitlines()[0]  # Keep the macro name and parameters, or its first line as a fallback.
46    if re.match(r"enum(?:\s+class)?\s+\w+", remainder):  # Keep the complete value list when the documented declaration is an enum.
47        end = remainder.find(";")  # Locate the declaration or directive end.
48        if 0 <= end < 800:  # Accept a nearby enum terminator while avoiding an unrelated later declaration.
49            return remainder[: end + 1]  # Include the enum declaration through its final semicolon.
50    lines = []  # Collect declaration signature lines.
51    for line in remainder.splitlines()[:14]:  # Limit the search to fourteen lines after the docstring.
52        line = line.split("//", 1)[0].rstrip()  # Remove inline C++ comments from the displayed signature.
53        found = re.search(r"[;{]", line)  # Locate the first declaration terminator on this line.
54        if found:  # Stop at a declaration terminator or the opening of an inline function body.
55            lines.append(line[: found.start()] + (";" if line[found.start()] == ";" else ""))  # Keep this line in the displayed C++ signature.
56            break  # Stop once the declaration boundary has been found.
57        if line.strip():  # Retain nonempty signature lines before that boundary.
58            lines.append(line)  # Keep this line in the displayed C++ signature.
59    return "\n".join(lines).strip()  # Return the collected signature with its original line breaks.

62def choose_group(declaration: str, comment: str, position: int, source: str) -> str:  # Place each API entry in a useful documentation section.
63    """Group public types and functions, with a GLSL namespace fallback."""

66        return GROUPS[0]  # Show version macros before the shader declaration interface.
67    if declaration.startswith("#define"):  # Keep declaration macros together even when their prose mentions resources.
68        return GROUPS[7]  # Use the adapter macro section.
69    if re.search(r"\b(headless_context|headless_device|backend_unavailable)\b", combined):  # Recognize the optional EGL public types by name.
70        return GROUPS[5]  # Use the EGL section.
71    async_start = source.find("// ---- async.hpp ----")  # Locate the asynchronous runtime section.
72    egl_start = source.find("// ---- egl.hpp ----")  # Locate the optional EGL section.
73    macros_start = source.find("// ---- macros.hpp ----")  # Locate the adapter declaration section.
74    if egl_start >= 0 and position >= egl_start and (macros_start < 0 or position < macros_start):  # Place declarations within the EGL source section in its own group.
75        return GROUPS[5]  # Use the EGL section.
76    if async_start >= 0 and position >= async_start and (egl_start < 0 or position < egl_start):  # Place declarations within the asynchronous source section in its own group.
77        return GROUPS[4]  # Use the asynchronous section.
78    if re.search(  # Recognize pipeline and registered factory declarations.
79        r"\b(pipeline|module|plugin|kernel_export|shared_library|load_kernel|make_kernel|register_kernel)\b",  # Recognize pipeline composition and dynamic factory names.
80        combined,  # Search the signature together with its explanation.
81    ):  # Use composition and factory names to choose the pipeline group.
82        return GROUPS[3]  # Use the pipeline section.
83    if re.search(r"\b(stage|dispatch|backend|kernel)\b", declaration):  # Recognize dispatch dimensions, stages and kernel interfaces.
84        return GROUPS[2]  # Use the kernel section.
85    if re.search(  # Recognize resource types and transfer methods.
86        r"\b(buffer|image2d|access|upload|download|from_gl|host_dirty|gpu_dirty)\b", combined  # Match the supported names in this API group.
87    ):  # Group buffer and image synchronization APIs together.
88        return GROUPS[1]  # Use the resource section.
89    if re.search(  # Recognize basic setup, source export and error declarations.
90        r"\b(device|profile|glsl_source|glsl_prelude|read_file|error|resolver|extent|ceil_div)\b",  # Recognize device setup, shader export and basic runtime names.
91        combined,  # Search the signature together with its explanation.

95    runtime_start = source.find("namespace shdr", glsl_start + 1)  # Find where runtime declarations follow the math section.

100    ):  # Keep declarations inside the math namespace in the math group.
101        return GROUPS[6]  # Use the math section.
102    return GROUPS[8]  # Use the remaining public declarations section.

105def family_symbols(body: str) -> list[str]:  # List functions covered by one math-family docstring.
106    """Collect names from public declarations and the header's math generator calls."""
107    body = re.sub(r"/\*[\s\S]*?\*/|//[^\n]*", "", body)  # Remove comments before scanning math declarations.
108    names = []  # Collect the public names found in this family.
109    depth = 0  # Track nested bodies so only public names are listed.
110    in_macro = False  # Track whether this line continues a macro definition.
111    for line in body.splitlines():  # Read each declaration and generator call in source order.
112        stripped = line.strip()  # Inspect the current line after trimming indentation.
113        if in_macro or stripped.startswith("#define"):  # Select public declarations surrounding the macro definitions.
114            in_macro = stripped.endswith("\\")  # Track whether this line continues a macro definition.
115            continue  # Skip this item and inspect the next one.
116        macro = re.match(r"SHDR_DETAIL_\w+\(([^,)]+)", stripped)  # Recognize a generated macro or declaration name.
117        if macro and depth == 0:  # Read a public symbol passed to a top-level math generator.
118            name = macro.group(1).strip()  # Trim surrounding whitespace from the symbol name.
119            if re.fullmatch(r"[A-Za-z_]\w*", name):  # Accept ordinary identifier names from a generator call.
120                names.append(name)  # Keep this public function or operator name.
121            elif re.fullmatch(r"[+*/%&|^<>=!-]+", name):  # Handle this source form.
122                names.append("operator" + name)  # Recognize operator tokens supplied to the math generator.
123        elif depth == 0 and not stripped.startswith("#"):  # Handle this source form.
124            candidates = re.findall(r"(?<![:\w])([A-Za-z_]\w*)\s*\(", line)  # Scan ordinary top-level declarations outside preprocessor directives.
125            candidates = [  # Find possible function names and remove language keywords.
126                name  # Keep this candidate function name.
127                for name in candidates  # Inspect the names matched on this declaration line.
128                if name  # Keep candidate public function names.
129                not in {"if", "for", "while", "switch", "static_assert", "decltype", "sizeof"}  # Remove language constructs that resemble function calls.
130            ]  # Remove language keywords from the list of possible function names.
131            if candidates:  # Keep the first public function name on this declaration line.
132                names.append(candidates[0])  # Keep this public function or operator name.
133        depth += line.count("{") - line.count("}")  # Track nested bodies so only public names are listed.
134    return list(dict.fromkeys(names))  # Return unique public names in source order.

137def documented_declarations(source: str) -> list[dict[str, str | int]]:  # Collect public declarations and their source locations.
138    """Collect header docstrings, declarations, groups, and source line numbers."""
139    expression = re.compile(r"/\*\*[\s\S]*?\*/|(?:^[ \t]*///[^\n]*(?:\n|$))+", re.MULTILINE)  # Match block docstrings and triple-slash comments.
140    entries = []  # Collect the documented public declarations.
141    for match in expression.finditer(source):  # Visit each matched public docstring.
142        comment = clean_comment(match.group())  # Read the cleaned C++ comment text.
143        if not comment or "@internal" in comment or "\\internal" in comment:  # Skip empty comments and entries explicitly marked internal.
144            continue  # Continue to the next public docstring.
145        if re.fullmatch(r"(?:@|\\)[{}]\s*", comment):  # Recognize structural Doxygen group markers.
146            continue  # Continue to the next documented declaration.
147        family = re.search(r"(?:@|\\)name\s+([^\n]+)", comment)  # Recognize a named group of math functions.
148        if family:  # Treat a named Doxygen family as one math API entry.
149            closing = re.search(r"/\*\*\s*(?:@|\\)}\s*\*/", source[match.end() :])  # Locate the end of the documented math family.
150            family_end = match.end() + closing.start() if closing else match.end()  # Limit the symbol scan to this family.
151            names = family_symbols(source[match.end() : family_end])  # Collect the public names found in this family.
152            declaration = family.group(1) + ("\n" + ", ".join(names) if names else "")  # Read the displayed signature for this API entry.
153            entries.append(  # Store the public signature, explanation, group and source line.
154                {  # Store the location needed to link this entry to its header.
155                    "comment": comment,  # Keep the entry's explanatory text.
156                    "declaration": declaration,  # Keep the displayed C++ signature.
157                    "group": GROUPS[6],  # Choose the API section for this entry.
158                    "line": source.count("\n", 0, match.start()) + 1,  # Link this entry to its header source line.
159                }  # Complete this family entry with its source location.
160            )  # Add the entire documented family as one API card.
161            continue  # Continue after publishing this complete family entry.
162        declaration = declaration_after(source, match.end())  # Read the displayed signature for this API entry.
163        if not declaration or re.match(r"(?:#(?:endif|if|include)|namespace\b)", declaration):  # Skip unattached comments and namespace or conditional directives.
164            continue  # Continue to the next attached public declaration.
165        entries.append(  # Store the public signature, explanation, group and source line.
166            {  # Store the location needed to link this entry to its header.
167                "comment": comment,  # Keep the entry's explanatory text.
168                "declaration": declaration,  # Keep the displayed C++ signature.
169                "group": choose_group(declaration, comment, match.start(), source),  # Choose the API section for this entry.
170                "line": source.count("\n", 0, match.start()) + 1,  # Link this entry to its header source line.
171            }  # Complete the ordinary declaration entry.
172        )  # Add this declaration to the exported API entries.
173    return entries  # Return all collected public API entries.

176def render_comment(comment: str) -> str:  # Convert the supported Doxygen tags to escaped HTML.
177    """Render a small Doxygen tag subset, escaping every source-content string."""
178    comment = re.sub(r"(?:@|\\)brief\s+", "", comment)  # Remove the brief tag while retaining its explanation.
179    comment = re.sub(r"(?:@|\\)(?:ingroup|defgroup)\s+[^\n]+", "", comment)  # Drop group-control tags from the displayed prose.
180    comment = re.sub(r"(?:@|\\)name\s+[^\n]+", "", comment)  # Keep the family title in its signature.
181    comment = re.sub(r"(?:@|\\)[{}]", "", comment)  # Remove Doxygen group-boundary markers from the prose.
182    paragraphs = []  # Collect escaped HTML paragraphs.
183    for paragraph in re.split(r"\n\s*\n", comment.strip()):  # Render each comment paragraph separately.
184        escaped = html.escape(paragraph)  # Escape text before adding supported formatting tags.
185        escaped = re.sub(  # Turn a parameter tag into a label while retaining its explanation.
186            r"(?:@|\\)param(?:\[[^]]+\])?\s+(\w+)\s+",  # Match an optional direction qualifier and the parameter name.
187            r"<strong>Parameter <code>\1</code>:</strong> ",  # Show each parameter name beside its explanation.
188            escaped,  # Keep the escaped explanation in the displayed paragraph.
189        )  # Finish rendering this parameter label.
190        for tags, label in (  # Handle each supported Doxygen label.
191            ("return|returns", "Returns"),  # Label documented return values.
192            ("throw|throws|exception", "Throws"),  # Label documented exceptions.
193            ("note", "Note"),  # Label additional usage notes.
194            ("warning", "Requirement"),  # Label required preconditions.
195        ):  # Apply each label to the matching Doxygen tag.
196            escaped = re.sub(  # Replace the current tag with a readable HTML label.
197                r"(?:@|\\)(?:" + tags + r")\s+", "<strong>" + label + ":</strong> ", escaped  # Replace this documented tag and preserve surrounding text.
198            )  # Keep the rest of this escaped paragraph unchanged.
199        escaped = re.sub(r"`([^`]+)`", r"<code>\1</code>", escaped)  # Render backtick-delimited text as inline code.
200        paragraphs.append("<p>" + escaped.replace("\n", "<br>") + "</p>")  # Keep this escaped paragraph in its original order.
201    return "\n".join(paragraphs)  # Return the joined HTML paragraphs.
