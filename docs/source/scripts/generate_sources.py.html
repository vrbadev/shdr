54def write_source(path: Path) -> Path:  # Publish one exact source file as readable HTML.

59    text = payload.decode("utf-8")  # All selected source files use UTF-8 text.

61    download = base64.b64encode(payload).decode("ascii")  # Encode the original file as a browser download.

66    body = '<p class="eyebrow">Source browser</p><h1><code>' + html.escape(relative.as_posix()) + '</code></h1><p>Read the complete file below or download its original bytes.
