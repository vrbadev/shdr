13def documentation_version():  # Identify the library version and preserve recorded report fields.
14    header = Path(__file__).resolve().parent.parent / "shdr.h"  # Locate the runtime documented by this checkout.
15    match = re.search(r'^\s*#define\s+SHDR_VERSION\s+"([^"]+)"', header.read_text(encoding="utf-8"), re.MULTILINE)  # Read the header's public version string.
16    if not match:  # Require an identifiable library version in the report title.
17        raise ValueError("SHDR_VERSION is missing from shdr.h")  # Identify the missing version declaration.
18    return escape(match.group(1))  # Escape the library version for HTML.

82def scalar(value, path):  # Render one exact report value with a machine-checkable original path.
83    kind = "null" if value is None else "boolean" if isinstance(value, bool) else "number" if isinstance(value, (int, float)) else "string"  # Preserve the original scalar category.
84    text = "null" if value is None else "true" if value is True else "false" if value is False else str(value)  # Format booleans and numeric values according to their recorded types.
85    attrs = f'data-report-path="{escape(path, quote=True)}" data-report-type="{kind}"'  # Identify this value independently of its presentation.
86    if text.startswith(("https://", "http://")) and "\n" not in text:  # Keep recorded source and license URLs directly usable.
87        return f'<a {attrs} href="{escape(text, quote=True)}">{escape(text)}</a>'  # Show the complete source URL and its provenance.
88    tag = "pre" if "\n" in text or path.endswith(".diagnostic") else "span"  # Keep diagnostic line breaks available to readers and preservation checks.
89    return f'<{tag} {attrs}>{escape(text)}</{tag}>'  # Escape compiler output before placing it in the document.

92def fields(value, path):  # Render nested report dictionaries and lists with every field retained.
