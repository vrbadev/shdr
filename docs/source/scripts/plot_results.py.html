13class ReportTables(HTMLParser):  # Collect the explicit publication tables and exact corpus scalar fields.
14    def __init__(self):  # Initialize the active table and captured field state.
15        super().__init__(convert_charrefs=True)  # Decode visible HTML entities back into their recorded text.
16        self.tables, self.values = {}, {}  # Retain table rows and path-addressed corpus values.
17        self.active, self.row, self.cell = None, None, None  # Track only the table currently being parsed.
18        self.row_attrs, self.capture = {}, None  # Retain optional row identity and one scalar capture.
19
20    def handle_starttag(self, tag, attrs):  # Recognize semantic table boundaries and report-value annotations.
21        attrs = dict(attrs)  # Access attributes by their names.
22        if tag == "table" and attrs.get("id"):  # Select tables carrying an authoritative identifier.
23            self.active = attrs["id"]  # Remember the current publication table.
24            self.tables[self.active] = []  # Collect its rows in displayed order.
25        if self.active and tag == "tr":  # Begin a row only within an identified table.
26            self.row, self.row_attrs = [], attrs  # Preserve visible cells and any stable row identity.
27        if self.row is not None and tag in ("td", "th"):  # Read both headings and ordinary data cells.
28            self.cell = []  # Begin accumulating this cell's visible text.
29        if attrs.get("data-report-path"):  # Read exact scalar provenance already visible in the corpus page.
30            self.capture = (tag, attrs["data-report-path"], [])  # Keep the closing tag and original dictionary path.
31
32    def handle_data(self, data):  # Preserve human-readable table text and recorded scalar values.
33        if self.cell is not None:  # Include nested code and link text in the current table cell.
34            self.cell.append(data)  # Keep text until the cell closes.
35        if self.capture is not None:  # Preserve the exact corpus value separately from surrounding table headings.
36            self.capture[2].append(data)  # Retain line breaks and scalar spelling.
37
38    def handle_endtag(self, tag):  # Finish each collected value at its semantic boundary.
39        if self.capture is not None and tag == self.capture[0]:  # Store a complete path-addressed report field.
40            self.values[self.capture[1]] = "".join(self.capture[2])  # Avoid changing diagnostics or hash strings.
41            self.capture = None  # Stop collecting after that scalar's closing tag.
42        if tag in ("td", "th") and self.cell is not None:  # Finish one displayed table cell.
43            self.row.append(" ".join("".join(self.cell).split()))  # Normalize only table-layout whitespace.
44            self.cell = None  # Wait for the next cell.
45        if tag == "tr" and self.row is not None:  # Finish one complete publication row.
46            self.tables[self.active].append((self.row_attrs, self.row))  # Retain attributes and cells together.
47            self.row = None  # Stop collecting cells until another row begins.
48        if tag == "table":  # Leave the current identified table.
49            self.active = None  # Collect text inside the selected table cells.

58def numeric_rows(report, identifier, start):  # Skip headings and return only rows with numeric measured fields.
59    result = []  # Keep the original visible row order.
60    for attrs, cells in report.tables[identifier]:  # Visit each heading or data row.
61        try:  # Parse measurements from numeric cells.
62            values = [float(value) for value in cells[start:]]  # Read the exact rounded numbers published in the table.
63        except ValueError:  # Keep headings out of the numerical record.
64            continue  # Continue with the next displayed row.
65        result.append((attrs, cells[:start], values))  # Retain labels and row identities with their numerical values.
66    return result  # Use the recorded measurement values and timing ratios.

69def save_svg(figure, output, title, description):  # Export a self-contained scientific vector figure.
70    figure.savefig(output, format="svg", bbox_inches="tight", metadata={"Date": "2026-10-15", "Creator": "shdr publication plots", "Title": title})  # Preserve the recorded measurement date.
71    text = output.read_text(encoding="utf-8")  # Add screen-reader text to the exported SVG root.
72    from html import escape  # Keep plot descriptions safe inside XML.
73    description = f'<title id="plot-title">{escape(title)}</title><desc id="plot-description">{escape(description)}</desc>'  # Explain units and scope inside the downloadable artifact.
74    text = re.sub(r"(<svg\b[^>]*)(>)", r'\1 role="img" aria-labelledby="plot-title plot-description"\2' + description, text, count=1)  # Make the SVG usable independently of its HTML figure caption.
75    text = "\n".join(line.rstrip() for line in text.splitlines()) + "\n"  # Remove exporter padding while preserving the SVG's paths and text.
76    output.write_text(text, encoding="utf-8")  # Keep every plot as a portable standalone file.
77    print(output.name)  # Report the artifact generated by this invocation.

199def fixture_plot(fixtures, output, np, plt):  # Illustrate actual prepared inputs and an explicitly independent signal reference.
200    earth = np.frombuffer((fixtures / "earth_rgb.bin").read_bytes(), dtype=np.uint8).reshape(64, 64, 3)  # Read the exact prepared NASA RGB image.
201    ecg = np.frombuffer((fixtures / "signal_ecg.bin").read_bytes(), dtype="<f4").reshape(2, 2048)[0].astype(float)  # Read the channel-major MLII signal in millivolts.
202    digits = np.frombuffer((fixtures / "mnist_test.bin").read_bytes(), dtype=np.uint8).reshape(-1, 28, 28)  # Read the shipped official test-image prefix.
203    labels = np.frombuffer((fixtures / "mnist_labels.bin").read_bytes(), dtype=np.uint8)  # Retain the original digit labels.
204    indices = [int(np.flatnonzero(labels == value)[0]) for value in range(10)]  # Show the first supplied instance of each class.
205    offsets = np.arange(65) - 32  # Center the documented odd-length FIR design.
206    coefficients = 2 * (40 / 360) * np.sinc(2 * (40 / 360) * offsets) * np.hamming(65)  # Independently calculate the Hamming-sinc 40 Hz low-pass reference.
207    coefficients = (coefficients / coefficients.sum()).astype(np.float32).astype(float)  # Match stored coefficient precision while keeping the reference accumulation independent.
208    filtered = np.convolve(ecg, coefficients)[:len(ecg)]  # Apply the documented causal zero-prefix FIR convention.
209    time = np.arange(len(ecg)) / 360  # Express sample positions in seconds at the public recording's sample rate.
210    frequencies = np.fft.rfftfreq(len(ecg), 1 / 360)  # Express the real-spectrum bins in hertz.
211    figure = plt.figure(figsize=(14.5, 8.5))  # Combine image, waveform, spectrum and digit examples into one clear fixture plate.
212    grid = figure.add_gridspec(3, 3, height_ratios=[1, 1, .65], width_ratios=[1, 1.1, 1.1])  # Reserve a full-width final row for the ten digit classes.
213    axis = figure.add_subplot(grid[:2, 0])  # Display the supplied low-resolution image.
214    axis.imshow(earth, interpolation="nearest")  # Preserve the prepared RGB8 pixels.
215    axis.set_title("NASA Blue Marble · prepared 64 × 64 RGB", fontsize=11)  # Identify the source and prepared image size.
216    axis.axis("off")  # Avoid implying a geographic coordinate system for the RGB preview.
217    axis = figure.add_subplot(grid[0, 1:])  # Show the original ECG and independent filter reference together.
218    axis.plot(time, ecg, linewidth=.8, color="#7a8fa8", label="Prepared MLII")  # Plot the actual float32 waveform from the fixture.
219    axis.plot(time, filtered, linewidth=1, color="#0c8a79", label="Independent 65-tap, 40 Hz FIR reference")  # Clearly distinguish a computed reference from archived runtime output.
220    axis.set_xlim(0, 2.5)  # Show the original samples from an initial waveform interval.
221    axis.set_xlabel("Time (s)")  # State waveform time units.
222    axis.set_ylabel("Potential (mV)")  # Preserve the published ADC scaling.
223    axis.legend(frameon=False, fontsize=9, ncol=2)  # Identify source samples and reference filtering.
224    axis.grid(alpha=.2)  # Make waveform amplitude comparisons readable.
225    axis = figure.add_subplot(grid[1, 1:])  # Show the independent full-fixture real Fourier spectrum.
226    for signal, color, name in [(ecg, "#7a8fa8", "Prepared MLII"), (filtered, "#0c8a79", "Independent FIR reference")]:  # Use the same sources and colors as the waveform panel.
227        amplitude = np.abs(np.fft.rfft(signal)) / len(signal)  # Compute a correctly normalized real-signal amplitude spectrum.
228        amplitude[1:-1] *= 2  # Double only the interior positive-frequency bins.
229        axis.semilogy(frequencies, np.maximum(amplitude, 1e-9), color=color, linewidth=.9, label=name)  # Use an explicit small display floor on the logarithmic amplitude axis.
230    axis.set_xlabel("Frequency (Hz)")  # Use physical frequency units on the axis.
231    axis.set_ylabel("Single-sided amplitude (mV)")  # State the spectrum normalization and signal unit.
232    axis.set_xlim(0, 180)  # Show the original 360 Hz recording's complete Nyquist range.
233    axis.grid(alpha=.2)  # Help locate the low-pass attenuation region.
234    digit_grid = grid[2, :].subgridspec(1, 10)  # Keep one original labeled sample per digit class.
235    for digit, index in enumerate(indices):  # Preserve an explicit reproducible sample-selection rule.
236        axis = figure.add_subplot(digit_grid[0, digit])  # Give each test digit its own clean preview.
237        axis.imshow(digits[index], cmap="gray", vmin=0, vmax=255, interpolation="nearest")  # Display the original unsigned-byte image values.
238        axis.set_title(f"Label {digit}\nTest index {index}", fontsize=9)  # Distinguish ground-truth labels from model predictions.
239        axis.axis("off")  # Keep pixel previews uncluttered.
240    figure.suptitle("Recorded public inputs · images, ECG and official handwritten digits", fontsize=16)  # Describe the figure as actual input data and reference signal analysis.
241    figure.tight_layout(rect=(0, 0, 1, .95))  # Separate the title from the specimen panels.
242    save_svg(figure, output, "Actual prepared public fixtures", "Actual NASA image pixels, PhysioNet MLII samples at 360 Hz, independently calculated 40 Hz FIR and Fourier references, and first shipped MNIST instance of each class. This is an input illustration, not an invented runtime measurement.")  # Keep source and reference distinctions with the standalone artifact.
243    plt.close(figure)  # Release plotting resources.
