100def prepare_cache(cache, fetch):  # Check each pinned repository before reading its sources.
101    """Read pinned snapshots. Download them only when --fetch is supplied."""
102    cache.mkdir(parents=True, exist_ok=True)  # Create the cache parent and preserve existing snapshots.
103    for repository in REPOSITORIES:  # Visit every pinned public source collection.
104        folder = cache / repository["id"]  # Locate the cached upstream repository.
105        if not (folder / ".git").exists():  # Create a Git snapshot for a fresh cache.
106            if not fetch:  # Require explicit download permission through the fetch option.
107                raise RuntimeError("missing " + repository["id"] + " cache. Run with --fetch first")  # Explain how to populate a missing pinned cache.
108            if folder.exists() and any(folder.iterdir()):  # Avoid placing a repository into an unrelated nonempty directory.
109                raise RuntimeError("cache destination is not an empty repository: " + str(folder))  # Protect unrelated files in a nonempty cache destination.
110            folder.mkdir(parents=True, exist_ok=True)  # Create the empty cache directory before Git uses it.
111            checked(["git", "init", "--quiet", str(folder)])  # Initialize an empty repository for this missing cache.
112            checked(["git", "remote", "add", "origin", repository["url"]], cwd=folder)  # Record the upstream URL as this repository's origin.
113            checked(  # Fetch the pinned commit with bounded download history.
114                [  # List the Git arguments for the pinned source download.
115                    "git",  # Use Git for the pinned source download.
116                    "fetch",  # Download the chosen revision before sparse checkout.
117                    "--depth=1",  # Avoid downloading unrelated commit history.
118                    "--filter=blob:none",  # Delay file downloads until the selected paths are checked out.
119                    "origin",  # Read from the recorded upstream repository.
120                    repository["revision"],  # Fetch the exact pinned commit.
121                ],  # Complete the arguments for downloading only the pinned revision.
122                cwd=folder,  # Run this Git operation inside its cache repository.
123            )  # Fetch inside this upstream cache repository.
124            checked(["git", "sparse-checkout", "init", "--cone"], cwd=folder)  # Enable sparse checkout to limit cached source files.
125            checked(["git", "sparse-checkout", "set"] + repository["paths"], cwd=folder)  # Select only the repository paths listed for this collection.
126            checked(["git", "checkout", "--quiet", "--detach", repository["revision"]], cwd=folder)  # Check out the pinned commit in detached mode.
127        revision = checked(["git", "rev-parse", "HEAD"], cwd=folder)  # Read the revision currently checked out in the cache.
128        if revision != repository["revision"]:  # Require the cache to match its pinned commit.
129            raise RuntimeError(  # Require the recorded snapshot revision.
130                repository["id"] + " cache is at a different revision. Use a fresh cache directory"  # Identify the cache revision validation error.
131            )  # Name the mismatched cache in the failure message.
132        if checked(["git", "status", "--porcelain", "--untracked-files=no"], cwd=folder):  # Reject modified tracked inputs before compiler checks begin.
133            raise RuntimeError(repository["id"] + " cache contains modified tracked files")  # Reject tracked edits that would change the recorded upstream inputs.

160        text = text.replace(str(original), replacement)  # Replace private paths while keeping the compiler explanation.
161    lines = [  # Collect only the most useful diagnostic lines.
162        line.strip()  # Remove diagnostic indentation before storing it.
163        for line in text.splitlines()  # Inspect each compiler output line for useful errors.
164        if "error" in line.lower() or "ERROR" in line or "unsupported" in line.lower()  # Keep the compiler's substantive error messages.
165    ]  # Keep the selected error lines in their original order.
166    return "\n".join(lines[:4])[:1200] if lines else text.strip()[:500]  # Return a short portable explanation.

337def summary(records):  # Count accepted and rejected sources for the report.
338    counts = Counter()  # Count outcomes across the complete collection.
339    repositories = {}  # Keep a separate tally for each upstream project.
340    reasons = Counter()  # Count frontend rejection categories.
341    for record in records:  # Count each source's compiler outcomes.
342        count = repositories.setdefault(record["repository"], Counter())  # Select this repository's outcome counters.
343        for tally in (counts, count):  # Update the collection total and this repository total.
344            tally["sources"] += 1  # Count this compiler outcome for the current total.
345            if record["upstream_reference_has_errors"]:  # Count the upstream fixtures intentionally associated with errors.
346                tally["upstream_reference_has_errors"] += 1  # Count this compiler outcome for the current total.
347            if record["opengl"]["status"] == "passed":  # Count original sources accepted in OpenGL mode.
348                tally["opengl_passed"] += 1  # Count this compiler outcome for the current total.
349                if not record["upstream_reference_has_errors"]:  # Separate accepted fixtures from upstream negative-test cases.
350                    tally["opengl_passed_without_reference_errors"] += 1  # Count this compiler outcome for the current total.
351                if record["cpu"]["status"] == "compiled":  # Count generated adapters accepted by standard C++11.
352                    tally["opengl_cpu_compiled"] += 1  # Count this compiler outcome for the current total.
353            elif record["vulkan"]["status"] == "passed":  # Handle this source form.
354                tally["vulkan_only_passed"] += 1  # Count this compiler outcome for the current total.
355            else:  # Count sources rejected by both reference compilation paths.
356                tally["reference_rejected"] += 1  # Count this compiler outcome for the current total.
357            tally["cpu_" + record["cpu"]["status"]] += 1  # Count this compiler outcome for the current total.
358            if record["cpu"]["status"] == "compiled":  # Count generated adapters accepted by standard C++11.
359                for profile, result in record["generated_gpu"].items():  # Count successful GPU profile checks.
360                    if result["status"] == "passed":  # Count this generated GPU profile only when its compiler passed.
361                        tally["cpu_and_" + profile + "_compiled"] += 1  # Count this compiler outcome for the current total.
362                if all(result["status"] == "passed" for result in record["generated_gpu"].values()):  # Count universal language coverage only when all five profiles passed.
363                    tally["cpu_and_all_profiles_compiled"] += 1  # Count this compiler outcome for the current total.
364        if record["cpu"]["status"] == "frontend_rejected":  # Count the reason when the adapter frontend rejects a construct.
365            reasons[record["cpu"]["category"]] += 1  # Count this frontend rejection category.
366    return {  # Return the collection, repository and rejection totals.
367        "counts": dict(counts),  # Store totals across all selected sources.
368        "repositories": {name: dict(count) for name, count in repositories.items()},  # Record the pinned source collections.
369        "frontend_rejection_categories": dict(reasons),  # Store the resource and language rejection categories.
370    }  # Return all counters together with the original source records.
